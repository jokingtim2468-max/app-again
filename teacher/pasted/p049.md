Plan:

1. Create a server-side `RemoteEvent` to allow clients to request the VIP gamepass purchase prompt.
2. Verify gamepass ownership securely using `MarketplaceService:UserOwnsGamePassAsync` when the player joins.
3. Listen to `PromptGamePassPurchaseFinished` to instantly grant VIP status attributes (enabling 2x coin multipliers and chat tags) upon successful purchase.

```lua
--!strict
-- Script (ServerScriptService/VIPShopServer)

local MarketplaceService = game:GetService("MarketplaceService")
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local VIP_GAMEPASS_ID = 12345678 -- Replace with your actual GamePass ID

local promptRemote = Instance.new("RemoteEvent")
promptRemote.Name = "PromptVIP"
promptRemote.Parent = ReplicatedStorage

promptRemote.OnServerEvent:Connect(function(player: Player)
	local success, err = pcall(function()
		MarketplaceService:PromptGamePassPurchase(player, VIP_GAMEPASS_ID)
	end)
	if not success then
		warn("Could not prompt purchase: " .. tostring(err))
	end
end)

Players.PlayerAdded:Connect(function(player: Player)
	local leaderstats = player:WaitForChild("leaderstats") :: Folder
	local coins = leaderstats:FindFirstChild("Coins") :: IntValue?
	
	local success, hasPass = pcall(function()
		return MarketplaceService:UserOwnsGamePassAsync(player.UserId, VIP_GAMEPASS_ID)
	end)
	
	if success and hasPass then
		player:SetAttribute("IsVIP", true)
	end
end)

MarketplaceService.PromptGamePassPurchaseFinished:Connect(function(player: Player, gamePassId: number, wasPurchased: boolean)
	if gamePassId == VIP_GAMEPASS_ID and wasPurchased then
		player:SetAttribute("IsVIP", true)
		print(player.Name .. " successfully acquired VIP status!")
	end
end)
```

Explanation: This server script handles gamepass purchasing securely by utilizing `MarketplaceService` to prompt the buyer and verify ownership both on join and via purchase completion events. It assigns a player attribute (`IsVIP`) which server-side scripts can reference to apply 2x coin multipliers and chat tags safely without trusting client-side state.
