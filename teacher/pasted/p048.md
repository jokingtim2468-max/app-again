`p048`
Plan:

1. Create a `RemoteFunction` in `ReplicatedStorage` to receive purchase requests from the client.
2. Define item costs and metadata in a server-side catalog table.
3. On server invoke, validate player `leaderstats.Coins`, deduct the exact cost, and clone the corresponding tool from `ServerStorage` into the player's Backpack.

```lua
--!strict
-- Script (ServerScriptService/ShopServer)

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local ServerStorage = game:GetService("ServerStorage")
local Players = game:GetService("Players")

local shopFunction = Instance.new("RemoteFunction")
shopFunction.Name = "BuyShopItem"
shopFunction.Parent = ReplicatedStorage

type ShopItem = {
	Cost: number,
	ToolName: string,
}

local catalog: {[string]: ShopItem} = {
	SpeedCoil = {Cost = 100, ToolName = "SpeedCoil"},
	GravityCoil = {Cost = 150, ToolName = "GravityCoil"},
}

shopFunction.OnServerInvoke = function(player: Player, itemName: string): boolean
	local itemInfo = catalog[itemName]
	if not itemInfo then
		return false
	end
	
	local leaderstats = player:FindFirstChild("leaderstats")
	if not leaderstats then
		return false
	end
	
	local coinsVal = leaderstats:FindFirstChild("Coins")
	if not coinsVal or not coinsVal:IsA("IntValue") then
		return false
	end
	
	if coinsVal.Value < itemInfo.Cost then
		return false
	end
	
	local templateTool = ServerStorage:FindFirstChild(itemInfo.ToolName)
	if not templateTool or not templateTool:IsA("Tool") then
		return false
	end
	
	-- Deduct currency and award tool securely on the server
	coinsVal.Value -= itemInfo.Cost
	
	local clonedTool = templateTool:Clone()
	clonedTool.Parent = player.Backpack
	
	return true
end
```

Explanation: This server-authoritative script validates every purchase request by verifying the item's existence, checking the player's `Coins` value, deducting the correct amount, and safely awarding the tool from `ServerStorage` to prevent any client-side exploits or currency duplication.
